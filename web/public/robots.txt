# Noord Karavane robots policy: search and live answers yes, model training no.
# In production Cloudflare puts its managed block (training crawlers disallowed) above this file.
# Groups for the same user agent merge (RFC 9309), so the lines below agree with that block and still
# hold if the managed block is ever switched off.

# Search indexes and live-answer fetchers that do not train models: allowed.

User-agent: ChatGPT-User
User-agent: Claude-SearchBot
User-agent: Claude-User
User-agent: PerplexityBot
Content-Signal: search=yes,ai-input=yes,ai-train=no
Allow: /

# Model-training crawlers: not allowed.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: anthropic-ai
User-agent: CCBot
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: meta-externalagent
User-agent: Bytespider
Disallow: /

# Everyone else, including Googlebot and Bingbot.
User-agent: *
Allow: /
